## 2. Guttersize Mantığı - Kritik Açıklama

**Soru:** Serbest formda "kutucuklar arası boşluk" nasıl çalışacak?

**Çözüm:** Guttersize iki şekilde kullanılacak:

### A. İlk Açılışta (Otomatik Grid)
Tüm uygulamalar sıralanırken aralarına otomatik %1 boşluk bırakılır:
```
[Kutu1] %1 [Kutu2] %1 [Kutu3]
  ↓        ↓        ↓
 %10     %10      %10
```

### B. Düzenleme Modunda
Kullanıcı bir kutucuğu hareket ettirdiğinde:
- Eğer başka bir kutucuğa %1'den daha yakın bırakırsa → Otomatik olarak itilir
- Eğer %1'den fazla boşluk bırakırsa → O boşluk otomatik olarak `SPACER` widget'a dönüşür

```kotlin
fun handleWidgetMove(movedWidget: WidgetEntity, allWidgets: List<WidgetEntity>) {
    allWidgets.forEach { other ->
        val distance = calculateDistance(movedWidget, other)
        
        if (distance < settings.guttersize) {
            // Çok yakın, otomatik it
            pushAway(movedWidget, other)
        } else if (distance > settings.guttersize * 2) {
            // Çok uzak, araya SPACER ekle
            insertSpacerBetween(movedWidget, other)
        }
    }
}
```

### B. Render Motoru

```kotlin
class LauncherCanvasView(context: Context) : ViewGroup(context) {
    
    override fun onLayout(changed: Boolean, l: Int, t: Int, r: Int, b: Int) {
        val screenWidth = r - l
        val screenHeight = b - t
        
        // 4 yönlü güvenli alan
        val paddingTop = (settings.safeAreaTop / 100f * screenHeight).toInt()
        val paddingBottom = (settings.safeAreaBottom / 100f * screenHeight).toInt()
        val paddingLeft = (settings.safeAreaLeft / 100f * screenWidth).toInt()
        val paddingRight = (settings.safeAreaRight / 100f * screenWidth).toInt()
        
        // Her widget'ı yerleştir
        widgets.forEach { widget ->
            val bounds = calculateWidgetBounds(widget, screenWidth, screenHeight)
            val view = findViewByWidgetId(widget.id)
            
            view.layout(
                bounds.left + paddingLeft,
                bounds.top + paddingTop,
                bounds.right + paddingLeft,
                bounds.bottom + paddingTop
            )
            
            // Spacer'ları düzenleme modunda göster/gizle
            if (widget.type == WidgetType.SPACER) {
                view.visibility = if (isEditMode) View.VISIBLE else View.GONE
            }
        }
    }
    
    override fun dispatchDraw(canvas: Canvas) {
        // Saf siyah arka plan
        canvas.drawColor(Color.BLACK)
        
        super.dispatchDraw(canvas)
        
        if (isEditMode) {
            drawRulers(canvas)
            drawSafeAreaBounds(canvas)
        }
    }
}
```

```kotlin
class FirstRunSetup {
    fun setupDefaultLayout(context: Context, apps: List<AppInfo>) {
        val settings = SettingsEntity(
            safeAreaTop = 3.0f,
            safeAreaBottom = 3.0f,
            safeAreaLeft = 3.0f,
            safeAreaRight = 3.0f,
            guttersize = 1.0f,
            defaultWidgetWidth = 10.0f,
            defaultWidgetHeight = 10.0f,
            unitType = UnitType.PERCENT,
            backgroundColor = "#000000",
            enableAnimations = true,
            animationSpeed = 1.0f
        )
        
        // Saat widget'ı (sol üst)
        val clockWidget = WidgetEntity(
            id = UUID.randomUUID().toString(),
            type = WidgetType.CLOCK,
            xValue = 5.0f,
            yValue = 5.0f,
            widthValue = 15.0f,
            heightValue = 5.0f,
            unitType = UnitType.PERCENT,
            zIndex = 0
        )
        
        // Uygulamaları grid şeklinde yerleştir (guttersize ile)
        var currentX = 5.0f
        var currentY = 15.0f
        var columnCount = 0
        val maxColumns = 8  // 8 kutu yan yana
        
        apps.forEach { app ->
            val widget = WidgetEntity(
                id = UUID.randomUUID().toString(),
                type = WidgetType.APP,
                packageName = app.packageName,
                xValue = currentX,
                yValue = currentY,
                widthValue = settings.defaultWidgetWidth,
                heightValue = settings.defaultWidgetHeight,
                unitType = settings.unitType,
                zIndex = 1
            )
            
            // Sonraki pozisyon (guttersize ile)
            currentX += settings.defaultWidgetWidth + settings.guttersize
            columnCount++
            
            if (columnCount >= maxColumns) {
                currentX = 5.0f
                currentY += settings.defaultWidgetHeight + settings.guttersize
                columnCount = 0
            }
        }
        
        // "Düzenleme Modu" butonu (sağ üst)
        val editModeWidget = WidgetEntity(
            id = UUID.randomUUID().toString(),
            type = WidgetType.EDIT_MODE_TOGGLE,
            xValue = 85.0f,
            yValue = 5.0f,
            widthValue = 10.0f,
            heightValue = 5.0f,
            unitType = UnitType.PERCENT,
            zIndex = 10
        )
        
        // "Başlatıcı Ayarları" butonu
        val settingsWidget = WidgetEntity(
            id = UUID.randomUUID().toString(),
            type = WidgetType.LAUNCHER_SETTINGS,
            xValue = 73.0f,
            yValue = 5.0f,
            widthValue = 10.0f,
            heightValue = 5.0f,
            unitType = UnitType.PERCENT,
            zIndex = 10
        )
        
        // Veritabanına kaydet
```

```kotlin
class EditModeOverlay(context: Context) : View(context) {
    
    override fun onDraw(canvas: Canvas) {
        val screenWidth = width.toFloat()
        val screenHeight = height.toFloat()
        
        // Üst cetvel
        for (i in 0..100 step 5) {
            val x = (i / 100f * screenWidth)
            canvas.drawLine(x, 0f, x, 20f, tickPaint)
            canvas.drawText("$i%", x, 40f, textPaint)
        }
        
        // Sol cetvel
        for (i in 0..100 step 5) {
            val y = (i / 100f * screenHeight)
            canvas.drawLine(0f, y, 20f, y, tickPaint)
            canvas.drawText("$i%", 30f, y, textPaint)
        }
        
        // Güvenli alan sınırları (kesikli kırmızı çizgi)
        val safeAreaLeft = (settings.safeAreaLeft / 100f * screenWidth)
        val safeAreaRight = screenWidth - (settings.safeAreaRight / 100f * screenWidth)
        val safeAreaTop = (settings.safeAreaTop / 100f * screenHeight)
        val safeAreaBottom = screenHeight - (settings.safeAreaBottom / 100f * screenHeight)
        
        canvas.drawRect(
            safeAreaLeft, safeAreaTop,
            safeAreaRight, safeAreaBottom,
            dashedPaint
        )
        
        // Spacer'ları göster
        widgets.filter { it.type == WidgetType.SPACER }.forEach { spacer ->
            val bounds = calculateWidgetBounds(spacer)
            canvas.drawRect(bounds, spacerPaint)
            canvas.drawText("SPACER", bounds.centerX(), bounds.centerY(), textPaint)
        }
    }
}
```
